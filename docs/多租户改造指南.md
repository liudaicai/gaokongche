## 🔧 阶段二：改造路由文件（核心）

### 改造模式（标准流程）

每个路由文件需要：
1. **导入中间件**
2. **应用中间件到路由**
3. **修改SQL查询添加WHERE过滤**
4. **INSERT/UPDATE时自动设置company_id**

---

### 示例：改造前 vs 改造后

#### ❌ 改造前（无数据隔离）

```javascript
// server/routes/orders.mysql.js
export default function buildOrdersRouter(pool) {
  const router = express.Router();
  
  // 查询所有订单（所有公司都能看到）
  router.get('/', async (req, res) => {
    const sql = `SELECT * FROM orders WHERE 1=1 ORDER BY created_at DESC`;
    const [rows] = await pool.query(sql);
    res.json({ ok: true, data: rows });
  });
  
  // 创建订单（没有设置company_id）
  router.post('/', async (req, res) => {
    const data = { ...req.body };
    const [result] = await pool.query('INSERT INTO orders SET ?', [data]);
    res.json({ ok: true, id: result.insertId });
  });
  
  return router;
}
```

#### ✅ 改造后（支持多租户）

```javascript
// server/routes/orders.mysql.js
import { tenantMiddleware, setTenantId } from '../middleware/tenant.js';

export default function buildOrdersRouter(pool) {
  const router = express.Router();
  
  // 应用中间件：自动过滤租户数据
  router.get('/', tenantMiddleware, async (req, res) => {
    const { where, params } = req.tenantFilter; // 中间件注入的过滤条件
    
    const sql = `
      SELECT * FROM orders 
      WHERE ${where}  -- 自动注入：company_id = ?
      ORDER BY created_at DESC
    `;
    
    const [rows] = await pool.query(sql, params);
    res.json({ ok: true, data: rows });
  });
  
  // 创建订单：自动设置company_id
  router.post('/', tenantMiddleware, async (req, res) => {
    const data = setTenantId(req, req.body); // 自动添加 company_id
    
    const [result] = await pool.query('INSERT INTO orders SET ?', [data]);
    res.json({ ok: true, id: result.insertId });
  });
  
  return router;
}
```

## ⚠️ 特殊场景处理

### 场景1：共享资源（如设备型号）

使用 `sharedResourceMiddleware` 允许访问系统级数据：

```javascript
import { sharedResourceMiddleware } from '../middleware/tenant.js';

// 设备型号：可以访问本公司+系统级（NULL）
router.get('/', sharedResourceMiddleware, async (req, res) => {
  const { where, params } = req.tenantFilter;
  // where = '(company_id = ? OR company_id IS NULL)'
  const sql = `SELECT * FROM equipment_models WHERE ${where}`;
  const [rows] = await pool.query(sql, params);
  res.json({ ok: true, data: rows });
});
```

### 场景2：关联表查询

确保关联表也要过滤：

```javascript
router.get('/', tenantMiddleware, async (req, res) => {
  const { where, params } = req.tenantFilter;
  
  const sql = `
    SELECT 
      o.*,
      c.name AS customer_name,
      e.name AS equipment_name
    FROM orders o
    LEFT JOIN customers c ON c.id = o.customer_id AND c.${where}
    LEFT JOIN equipments e ON e.id = o.equipment_id AND e.${where}
    WHERE o.${where}
    ORDER BY o.created_at DESC
  `;
  
  // 注意：params 需要重复3次（orders + customers + equipments）
  const [rows] = await pool.query(sql, [...params, ...params, ...params]);
  res.json({ ok: true, data: rows });
});
```

### 场景3：超级管理员绕过限制

某些API超级管理员需要看到所有数据：

```javascript
router.get('/admin/all', tenantMiddleware, async (req, res) => {
  // 超级管理员：req.tenantFilter = { where: '1=1', params: [] }
  // 普通用户：req.tenantFilter = { where: 'company_id = ?', params: [123] }
  
  const { where, params } = req.tenantFilter;
  const sql = `SELECT * FROM orders WHERE ${where}`;
  const [rows] = await pool.query(sql, params);
  res.json({ ok: true, data: rows });
});
```
